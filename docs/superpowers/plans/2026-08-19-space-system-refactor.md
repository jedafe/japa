# Space System Refactor Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Refactor JapaConnect spaces into a category + sub-type model, remove manual geo selection from space creation, auto-link spaces to primary geo spaces, and turn `Global` into a virtual geo-fallback feed.

**Architecture:** Pull-based `/feed/for-you` and `/feed/explore` are extended with a geo-proximity ranking (followed → city → state → country → region → global). The `space_types` table gains a `category` column; the member create flow becomes name + type only; the backend keyword-matches the name against primary spaces to populate `space_parent_spaces`. Admin `/space-types` uses a category selector.

**Tech Stack:** Go 1.26.4, sqlc, PostgreSQL 18, SvelteKit 2 + Svelte 5 Runes, bun, pgx/v5, chi.

## Global Constraints

- All migrations are `up.sql`/`down.sql` pairs using `golang-migrate`.
- Do not hand-edit `internal/db/gen/*`; edit `internal/db/queries/*.sql` and run `make sqlc-gen`.
- All JS/TS tooling uses `bun` (Rule 18); no `npm`, `npx`, `pnpm`, or `yarn`.
- `country`/`state`/`city` remain system-only; members cannot create them.
- `Global` is not a space or space type after the refactor.
- No AI-generated attribution in commits or code (Rule 17).
- Every migration must be idempotent (`ON CONFLICT DO NOTHING` or `IF NOT EXISTS`).
- All new Go error paths wrap with `fmt.Errorf("...: %w", err)`.

- [ ] **Step 1: Add a service method to auto-link a new space to primary spaces**

```go
// LinkSpaceToPrimarySpaces keyword-matches the space name against primary
// country/state/city spaces and inserts space_parent_spaces rows. It returns
// the most specific matched space ID (city > state > country) for parent_id.
func (s *Service) LinkSpaceToPrimarySpaces(ctx context.Context, spaceID pgtype.UUID, ID pgtype.UUID, name string) (pgtype.UUID, []pgtype.UUID, error) {
    keywords := extractSpaceKeywords(name)
    if len(keywords) == 0 {
        return pgtype.UUID{}, nil, nil
    }
    matches, err := s.FindSpacesByKeywords(ctx, ID, keywords)
    if err != nil {
        return pgtype.UUID{}, nil, fmt.Errorf("finding primary spaces by keywords: %w", err)
    }
    if len(matches) == 0 {
        return pgtype.UUID{}, nil, nil
    }

    // Track all matched IDs for space_parent_spaces.
    parentIDs := make([]pgtype.UUID, 0, len(matches))
    for _, m := range matches {
        var id pgtype.UUID
        if err := id.Scan(m.ID); err != nil { continue }
        parentIDs = append(parentIDs, id)
    }

    // Most specific match becomes parent_id.
    var primary pgtype.UUID
    _ = primary.Scan(matches[0].ID)
    return primary, parentIDs, nil
}
```

- [ ] **Step 2: Add `extractSpaceKeywords` helper in `internal/spaces/spaces.go`**

Reuse the tokenization from `internal/content/handlers.go` with a lower max n-gram length (2) for short names. Define a local stop-word map in `internal/spaces/spaces.go`:

```go
var spaceKeywordStopWords = map[string]struct{}{
    "a": {}, "an": {}, "the": {}, "in": {}, "of": {}, "for": {}, "and": {}, "or": {},
    "community": {}, "global": {}, "world": {}, "university": {}, "college": {},
}

var spaceKeywordSeparators = regexp.MustCompile(`[^\p{L}\p{N}]+`)

func extractSpaceKeywords(name string) []string {
    if name == "" {
        return nil
    }
    normalized := spaceKeywordSeparators.ReplaceAllString(strings.ToLower(strings.TrimSpace(name)), " ")
    raw := strings.Fields(normalized)
    words := make([]string, 0, len(raw))
    for _, w := range raw {
        if len(w) >= 2 {
            words = append(words, w)
        }
    }
    if len(words) == 0 {
        return nil
    }
    seen := make(map[string]struct{})
    out := make([]string, 0, len(words)*2)
    maxN := 2
    if len(words) < maxN {
        maxN = len(words)
    }
    for n := 1; n <= maxN; n++ {
        for i := 0; i+n <= len(words); i++ {
            phrase := strings.Join(words[i:i+n], " ")
            if n == 1 {
                if _, ok := spaceKeywordStopWords[phrase]; ok {
                    continue
                }
            }
            if _, ok := seen[phrase]; ok {
                continue
            }
            seen[phrase] = struct{}{}
            out = append(out, phrase)
        }
    }
    return out
}
```

- [ ] **Step 1: Replace `GetExploreFeed` with a geo-proximity query**

```sql
-- name: GetExploreFeed :many
-- Geo-aware public explore feed. Returns public content ordered by a blend
-- of geo-proximity to the visitor's country and popularity.
WITH visitor_country AS (
    SELECT id, region, geo_code
    FROM spaces
    WHERE tenant_id = @tenant_id
      AND level = 'country'
      AND (@visitor_country_code::text = '' OR geo_code = @visitor_country_code::text)
    LIMIT 1
)
SELECT c.id, c.title, c.slug, c.content_type, c.author_id, c.space_id,
       c.created_at, c.view_count, c.comment_count,
       cs.vote_score, cs.velocity_score,
       COALESCE(up.display_name, u.username, '') AS author_name,
       COALESCE(up.avatar_url, '') AS author_avatar_url,
       COALESCE(s.name, '') AS space_name,
       COALESCE(s.slug, '') AS space_slug,
       MAX(LEFT(c.body, 200))::text AS excerpt,
       cs.bookmark_count,
       c.read_time_minutes,
       m.storage_bucket AS cover_storage_bucket,
       m.storage_path AS cover_storage_path,
       COALESCE(string_agg(t.name, ',' ORDER BY t.name), '') AS tags,
       CASE
           WHEN s.level = 'city' AND s.parent_id IN (SELECT id FROM visitor_country) THEN 5
           WHEN s.level = 'city' AND sps.parent_space_id IN (SELECT id FROM visitor_country) THEN 5
           WHEN s.level = 'state' AND s.parent_id IN (SELECT id FROM visitor_country) THEN 4
           WHEN s.level = 'country' AND s.geo_code = @visitor_country_code::text THEN 3
           WHEN s.level = 'country' AND s.region = (SELECT region FROM visitor_country) THEN 2
           ELSE 1
       END AS geo_proximity
FROM contents c
JOIN content_scores cs ON cs.content_id = c.id
LEFT JOIN users u ON u.id = c.author_id
LEFT JOIN user_profiles up ON up.user_id = c.author_id
LEFT JOIN spaces s ON s.id = c.space_id
LEFT JOIN space_parent_spaces sps ON sps.space_id = s.id
LEFT JOIN content_tags ct ON ct.content_id = c.id
LEFT JOIN tags t ON t.id = ct.tag_id
LEFT JOIN media m ON m.id = COALESCE(c.featured_image_id, c.header_image_id) AND m.deleted_at IS NULL
WHERE c.tenant_id = @tenant_id
  AND c.status = 'published'
  AND c.deleted_at IS NULL
  AND c.visibility = 'public'
GROUP BY c.id, cs.vote_score, cs.velocity_score, cs.bookmark_count,
         u.username, up.display_name, up.avatar_url,
         s.name, s.slug, s.level, s.parent_id, s.geo_code, sps.parent_space_id,
         m.storage_bucket, m.storage_path
ORDER BY geo_proximity DESC, cs.velocity_score DESC, cs.vote_score DESC
LIMIT @page_limit OFFSET @page_offset;
```
